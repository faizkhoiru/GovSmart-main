% govsmart_fig3.tex — Standalone status-comprehension bar chart for GovSmart paper (N=31, 155 Trials)
% Horizontal stacked bars showing correct / partly correct / incorrect per status
% Compile: pdflatex govsmart_fig3.tex
\documentclass[border=4pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{xcolor}

\definecolor{cCorr}{HTML}{2E7D32}    % Correct — green
\definecolor{cPart}{HTML}{F9A825}    % Partly correct — amber
\definecolor{cInc}{HTML}{C62828}     % Incorrect — red
\definecolor{cAnnot}{HTML}{E65100}   % Annotation colour

\begin{document}
\begin{tikzpicture}
\begin{axis}[
    title={\textbf{Status Comprehension Accuracy across Operational Tokens} ($N{=}31$, 155 Trials)},
    title style={font=\small, at={(0.5,1.06)}, anchor=south},
    xbar stacked,
    width=14cm, height=6.5cm,
    bar width=14pt,
    xmin=0, xmax=115,
    xlabel={Comprehension Rate (\%)},
    xlabel style={font=\footnotesize},
    ytick={1,2,3,4,5},
    yticklabels={\textit{diterima}, \textit{ditolak}, \textit{diproses}, \textit{dibatalkan}, \textit{selesai}},
    yticklabel style={font=\footnotesize, anchor=east},
    tick label style={font=\footnotesize},
    legend style={
        at={(0.5,-0.18)},
        anchor=north,
        legend columns=3,
        font=\scriptsize,
        draw=none,
        column sep=6pt,
    },
    enlarge y limits={abs=0.55},
    axis lines*=left,
    clip=false,
]

% Correct: selesai (100%), dibatalkan (96.8%), diproses (87.1%), ditolak (83.9%), diterima (61.3%)
\addplot[fill=cCorr, draw=none] coordinates {
    (100.0,5) (96.8,4) (87.1,3) (83.9,2) (61.3,1)
};
% Partly correct: selesai (0%), dibatalkan (3.2%), diproses (12.9%), ditolak (16.1%), diterima (35.5%)
\addplot[fill=cPart, draw=none] coordinates {
    (0.0,5) (3.2,4) (12.9,3) (16.1,2) (35.5,1)
};
% Incorrect: diterima (3.2%)
\addplot[fill=cInc, draw=none] coordinates {
    (0.0,5) (0.0,4) (0.0,3) (0.0,2) (3.2,1)
};

% Percentage labels
\node[font=\scriptsize, white] at (axis cs:50,5) {100.0\%};
\node[font=\scriptsize, white] at (axis cs:48.4,4) {96.8\%};
\node[font=\scriptsize, white] at (axis cs:43.5,3) {87.1\%};
\node[font=\scriptsize, white] at (axis cs:42.0,2) {83.9\%};
\node[font=\scriptsize, white] at (axis cs:30.6,1) {61.3\%};

% Partly-correct labels (only where noticeable)
\node[font=\scriptsize] at (axis cs:93.5,3) {12.9\%};
\node[font=\scriptsize] at (axis cs:92.0,2) {16.1\%};
\node[font=\scriptsize] at (axis cs:79.0,1) {35.5\%};

% Ambiguity annotation — callout placed to the right, well clear of bars
\node[font=\scriptsize, cAnnot, text width=3.8cm, align=left, anchor=west]
    at (axis cs:105,1.8) {%
        \textbf{Ambiguity Point} ($n{=}11/31$):\\
        Token ``\textit{diterima}'' conflated\\
        with immediate officer dispatch.};

\legend{Correct, Partially Correct, Incorrect}
\end{axis}
\end{tikzpicture}
\end{document}
