% govsmart_fig2.tex — Standalone two-panel figure for GovSmart paper
% Panel (a): Stacked horizontal bar chart — task completion breakdown (N=31)
% Panel (b): Vertical bar chart — execution duration (median + range) (N=31)
\documentclass[border=6pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{xcolor}

% --- colour palette (forest / amber / brick) ---
\definecolor{cS}{HTML}{2E7D32}   % Success (S) — green
\definecolor{cSH}{HTML}{F9A825}  % Success with help (SH) — amber
\definecolor{cF}{HTML}{C62828}   % Failure (F) — red
\definecolor{cBar}{HTML}{1565C0} % Duration bars — blue
\definecolor{cTimeout}{HTML}{C62828} % Timeout line

\begin{document}
\begin{tikzpicture}
% =========================================================================
% === PANEL (A): Task Completion Outcomes ===
% =========================================================================
\begin{axis}[
    name=axA,
    at={(0cm, 0cm)},
    title={\textbf{(a) Task Completion Outcomes} ($N{=}31$)},
    title style={font=\footnotesize, yshift=2pt},
    xbar stacked,
    width=6.5cm, height=4.4cm,
    bar width=13pt,
    xmin=0, xmax=100,
    xlabel={Percentage (\%)},
    xlabel style={font=\scriptsize},
    ytick={1,2,3},
    yticklabels={T3 Find status, T2 Revise, T1 Submit},
    yticklabel style={font=\scriptsize, anchor=east},
    xticklabel style={font=\scriptsize},
    legend style={
        at={(0.5,-0.30)},
        anchor=north,
        legend columns=3,
        font=\scriptsize,
        draw=none,
        column sep=6pt,
    },
    enlarge y limits={abs=0.55},
    axis lines*=left,
    clip=false,
]
% S segments
\addplot[fill=cS, draw=none] coordinates {(93.5,1) (83.9,2) (71.0,3)};
% SH segments
\addplot[fill=cSH, draw=none] coordinates {(6.5,1) (16.1,2) (29.0,3)};
% F segments (all zero)
\addplot[fill=cF, draw=none] coordinates {(0,1) (0,2) (0,3)};

\legend{Success (S), Assisted (SH), Failure (F)}
\end{axis}

% =========================================================================
% === PANEL (B): Execution Duration (Median & Range) ===
% =========================================================================
\begin{axis}[
    name=axB,
    at={(8.6cm, 0cm)},
    title={\textbf{(b) Execution Duration (Median \& Range)}},
    title style={font=\footnotesize, yshift=2pt},
    ybar,
    width=6.5cm, height=4.4cm,
    bar width=16pt,
    ymin=0, ymax=350,
    ylabel={Time (seconds)},
    ylabel style={font=\scriptsize, yshift=2pt},
    xtick={1,2,3},
    xticklabels={T1 Submit, T2 Revise, T3 Status},
    xticklabel style={font=\scriptsize},
    yticklabel style={font=\scriptsize},
    enlarge x limits={abs=0.6},
    axis lines*=left,
    clip=false,
    ytick={0,50,100,150,200,250,300},
]
% Median bars: T1=138s, T2=75s, T3=38s
\addplot[fill=cBar, draw=none] coordinates {(1,138) (2,75) (3,38)};

% Error bars (range: min to max)
% T1: 95–260, median 138
\draw[thick, black] (axis cs:1,95) -- (axis cs:1,260);
\draw[thick, black] (axis cs:0.88,95) -- (axis cs:1.12,95);
\draw[thick, black] (axis cs:0.88,260) -- (axis cs:1.12,260);

% T2: 48–135, median 75
\draw[thick, black] (axis cs:2,48) -- (axis cs:2,135);
\draw[thick, black] (axis cs:0.88+1,48) -- (axis cs:1.12+1,48);
\draw[thick, black] (axis cs:0.88+1,135) -- (axis cs:1.12+1,135);

% T3: 20–65, median 38
\draw[thick, black] (axis cs:3,20) -- (axis cs:3,65);
\draw[thick, black] (axis cs:0.88+2,20) -- (axis cs:1.12+2,20);
\draw[thick, black] (axis cs:0.88+2,65) -- (axis cs:1.12+2,65);

% Median labels
\node[font=\scriptsize\bfseries, anchor=west] at (axis cs:1.18,138) {138\,s};
\node[font=\scriptsize\bfseries, anchor=west] at (axis cs:2.18,75) {75\,s};
\node[font=\scriptsize\bfseries, anchor=west] at (axis cs:3.18,38) {38\,s};

% Timeout dashed line at y=300 with label
\draw[cTimeout, dashed, thick] (axis cs:0.4,300) -- (axis cs:3.6,300);
\node[font=\scriptsize\bfseries, cTimeout, anchor=south east] at (axis cs:3.6,302) {Timeout = 300\,s};

\end{axis}
\end{tikzpicture}
\end{document}
